        "https://s3.amazonaws.com/data.patentsview.org/download/g_patent_assignee.tsv.zip",
}

def download(url, dest):
    if os.path.exists(dest):
        print(f"  skip (exists): {dest}"); return
    r = requests.get(url, stream=True)
    total = int(r.headers.get('content-length', 0))
    with open(dest, "wb") as fh, tqdm(total=total, unit="B", unit_scale=True, desc=os.path.basename(dest)) as bar:
        for chunk in r.iter_content(65536):
            fh.write(chunk); bar.update(len(chunk))

for fname, url in FILES.items():
    download(url, f"{RAW}/{fname}")

print("\nUnzipping...")
os.system(f"unzip -o 'data/raw/*.zip' -d data/raw/")

PHASE 2
~60 min
Clean Data with Chunked pandas
At millions of rows you CANNOT load the full TSV into memory at once. You must use pandas chunking — read, clean, and write in batches of 100k rows. This keeps memory use flat regardless of file size.
Never do this with big files: df = pd.read_csv("g_patent.tsv") — this will consume all your RAM and crash. Always use chunksize=.

Script: 02_clean.py — full chunked cleaning pipeline

Python — scripts/02_clean.py
import pandas as pd
import os
from tqdm import tqdm

RAW   = "data/raw"
CLEAN = "data/clean"
os.makedirs(CLEAN, exist_ok=True)

CHUNK = 100_000  # rows per chunk — tune based on your RAM


# ─── HELPER: count rows quickly without loading full file ──────────────────────
def count_lines(path):
    with open(path, "rb") as f:
        return sum(1 for _ in f) - 1  # subtract header


# ─── 1. PATENTS ────────────────────────────────────────────────────────────────
print("[1/5] Cleaning patents...")
out_path = f"{CLEAN}/clean_patents.csv"
header_written = False
total_rows = 0

reader = pd.read_csv(
    f"{RAW}/g_patent.tsv",
    sep="\t",
    usecols=["patent_id", "patent_title", "patent_abstract", "patent_date"],
    chunksize=CHUNK,
    low_memory=False,
    on_bad_lines="skip"  # skip malformed rows
)

for chunk in tqdm(reader, desc="patents"):
    chunk.rename(columns={
        "patent_title": "title",
        "patent_abstract": "abstract",
        "patent_date": "filing_date"
    }, inplace=True)

    chunk.dropna(subset=["patent_id", "title"], inplace=True)
    chunk.drop_duplicates(subset=["patent_id"], inplace=True)

    chunk["filing_date"] = pd.to_datetime(chunk["filing_date"], errors="coerce")
    chunk["year"]       = chunk["filing_date"].dt.year.astype("Int64")
    chunk["filing_date"] = chunk["filing_date"].dt.strftime("%Y-%m-%d")

    chunk["title"]    = chunk["title"].str.strip()
    chunk["abstract"] = chunk["abstract"].fillna("No abstract").str.strip()

    chunk[["patent_id", "title", "abstract", "filing_date", "year"]].to_csv(
        out_path, mode="a", index=False, header=not header_written
    )
    header_written = True
    total_rows += len(chunk)

print(f"  ✓ Patents: {total_rows:,} rows")


# ─── 2. INVENTORS ──────────────────────────────────────────────────────────────
print("[2/5] Cleaning inventors...")
out_path = f"{CLEAN}/clean_inventors.csv"
header_written = False; total_rows = 0

reader = pd.read_csv(
    f"{RAW}/g_inventor_disambiguated.tsv",
    sep="\t",
    usecols=["disambig_inventor_id_20", "disambig_inventor_name_first",
             "disambig_inventor_name_last", "disambig_country"],
    chunksize=CHUNK, low_memory=False, on_bad_lines="skip"
)

seen_ids = set()  # track dedup across chunks

for chunk in tqdm(reader, desc="inventors"):
    chunk.rename(columns={
        "disambig_inventor_id_20": "inventor_id",
        "disambig_country": "country"
    }, inplace=True)

    chunk["name"] = (
        chunk["disambig_inventor_name_first"].fillna("") + " " +
        chunk["disambig_inventor_name_last"].fillna("")
    ).str.strip()

    chunk = chunk[["inventor_id", "name", "country"]]
    chunk.dropna(subset=["inventor_id"], inplace=True)
    chunk["country"] = chunk["country"].fillna("Unknown")

    # Cross-chunk dedup
    chunk = chunk[~chunk["inventor_id"].isin(seen_ids)]
    seen_ids.update(chunk["inventor_id"].tolist())

    chunk.to_csv(out_path, mode="a", index=False, header=not header_written)
    header_written = True; total_rows += len(chunk)

print(f"  ✓ Inventors: {total_rows:,} rows")


# ─── 3. COMPANIES ──────────────────────────────────────────────────────────────
print("[3/5] Cleaning companies...")
out_path = f"{CLEAN}/clean_companies.csv"
header_written = False; total_rows = 0
seen_ids = set()

reader = pd.read_csv(
    f"{RAW}/g_assignee_disambiguated.tsv",
    sep="\t",
    usecols=["disambig_assignee_id_20", "disambig_assignee_organization"],
    chunksize=CHUNK, low_memory=False, on_bad_lines="skip"
)

for chunk in tqdm(reader, desc="companies"):
    chunk.rename(columns={
        "disambig_assignee_id_20": "company_id",
        "disambig_assignee_organization": "name"
    }, inplace=True)
    chunk.dropna(subset=["company_id", "name"], inplace=True)
    chunk["name"] = chunk["name"].str.strip()
    chunk = chunk[~chunk["company_id"].isin(seen_ids)]
    seen_ids.update(chunk["company_id"].tolist())
    chunk.to_csv(out_path, mode="a", index=False, header=not header_written)
    header_written = True; total_rows += len(chunk)

print(f"  ✓ Companies: {total_rows:,} rows")


# ─── 4 & 5. RELATIONSHIPS ──────────────────────────────────────────────────────
print("[4/5] Cleaning patent-inventor links...")
out_path = f"{CLEAN}/clean_patent_inventor.csv"
header_written = False; total_rows = 0

reader = pd.read_csv(f"{RAW}/g_patent_inventor.tsv", sep="\t",
                      usecols=["patent_id", "inventor_id"],
                      chunksize=CHUNK, low_memory=False)
for chunk in tqdm(reader, desc="pat-inv"):
    chunk.dropna(inplace=True)
    chunk.to_csv(out_path, mode="a", index=False, header=not header_written)
    header_written = True; total_rows += len(chunk)
print(f"  ✓ Pat-Inv links: {total_rows:,} rows")

print("[5/5] Cleaning patent-assignee links...")
out_path = f"{CLEAN}/clean_patent_assignee.csv"
header_written = False; total_rows = 0

reader = pd.read_csv(f"{RAW}/g_patent_assignee.tsv", sep="\t",
                      usecols=["patent_id", "assignee_id"],
                      chunksize=CHUNK, low_memory=False)
for chunk in tqdm(reader, desc="pat-asgn"):
    chunk.dropna(inplace=True)
    chunk.rename(columns={"assignee_id": "company_id"}, inplace=True)
    chunk.to_csv(out_path, mode="a", index=False, header=not header_written)
    header_written = True; total_rows += len(chunk)
print(f"  ✓ Pat-Assignee links: {total_rows:,} rows")

print("\n✓ All clean CSVs written to data/clean/")
Key Big Data technique: by writing each chunk to CSV with mode="a" (append), you produce the full cleaned output while never holding more than 100k rows in memory at once.
PHASE 3
~50 min
PostgreSQL Schema + Bulk Loading with COPY
Create the schema, then load millions of rows using PostgreSQL's COPY command — this is 50–100× faster than INSERT and is the standard production method for bulk loading.

File: sql/schema.sql

SQL — sql/schema.sql
-- Drop tables in reverse dependency order
DROP TABLE IF EXISTS patent_inventor  CASCADE;
DROP TABLE IF EXISTS patent_assignee  CASCADE;
DROP TABLE IF EXISTS patents          CASCADE;
DROP TABLE IF EXISTS inventors        CASCADE;
DROP TABLE IF EXISTS companies        CASCADE;

-- Core tables
CREATE TABLE patents (
    patent_id   TEXT        PRIMARY KEY,
    title       TEXT        NOT NULL,
    abstract    TEXT,
    filing_date DATE,
    year        SMALLINT
);

CREATE TABLE inventors (
    inventor_id TEXT        PRIMARY KEY,
    name        TEXT,
    country     TEXT
);

CREATE TABLE companies (
    company_id  TEXT        PRIMARY KEY,
    name        TEXT
);

-- Relationship tables (replaces single joined table for better performance)
CREATE TABLE patent_inventor (
    patent_id   TEXT REFERENCES patents(patent_id)  ON DELETE CASCADE,
    inventor_id TEXT REFERENCES inventors(inventor_id) ON DELETE CASCADE,
    PRIMARY KEY (patent_id, inventor_id)
);

CREATE TABLE patent_assignee (
    patent_id  TEXT REFERENCES patents(patent_id)  ON DELETE CASCADE,
    company_id TEXT REFERENCES companies(company_id) ON DELETE CASCADE,
    PRIMARY KEY (patent_id, company_id)
);

-- Indexes for big-data query performance
CREATE INDEX idx_patents_year     ON patents(year);
CREATE INDEX idx_inventors_country ON inventors(country);
CREATE INDEX idx_pat_inv_inventor  ON patent_inventor(inventor_id);
CREATE INDEX idx_pat_asgn_company  ON patent_assignee(company_id);

Python — scripts/03_load_db.py
import os, sys
import psycopg2
sys.path.insert(0, "scripts")
from config import get_conn   # our helper from 00_config.py

CLEAN = "data/clean"

# ─── 1. Create schema ──────────────────────────────────────────────────────────
print("Creating schema...")
conn = get_conn()
with conn, conn.cursor() as cur:
    with open("sql/schema.sql") as f:
        cur.execute(f.read())
print("  ✓ Schema ready")


# ─── 2. COPY helper — this is the fast path ────────────────────────────────────
def copy_csv(conn, csv_path, table, columns):
    """
    Uses PostgreSQL COPY FROM — reads directly from CSV file.
    This is the fastest way to load data: ~500k rows/second.
    """
    abs_path = os.path.abspath(csv_path)
    print(f"  COPY {table} ← {os.path.basename(csv_path)}")
    with conn, conn.cursor() as cur:
        cols = ", ".join(columns)
        sql = f"""
            COPY {table} ({cols})
            FROM '{abs_path}'
            WITH (FORMAT csv, HEADER true, NULL '', QUOTE '"')
        """
        cur.execute(sql)
        print(f"    ✓ {cur.rowcount:,} rows loaded")


# ─── 3. Load all tables in dependency order ────────────────────────────────────
conn = get_conn()

copy_csv(conn, f"{CLEAN}/clean_patents.csv",
         "patents", ["patent_id", "title", "abstract", "filing_date", "year"])

copy_csv(conn, f"{CLEAN}/clean_inventors.csv",
         "inventors", ["inventor_id", "name", "country"])

copy_csv(conn, f"{CLEAN}/clean_companies.csv",
         "companies", ["company_id", "name"])

copy_csv(conn, f"{CLEAN}/clean_patent_inventor.csv",
         "patent_inventor", ["patent_id", "inventor_id"])

copy_csv(conn, f"{CLEAN}/clean_patent_assignee.csv",
         "patent_assignee", ["patent_id", "company_id"])

conn.close()

# ─── 4. Quick row count verification ──────────────────────────────────────────
print("\nVerifying row counts...")
conn = get_conn()
with conn.cursor() as cur:
    for t in ["patents", "inventors", "companies", "patent_inventor", "patent_assignee"]:
        cur.execute(f"SELECT COUNT(*) FROM {t}")
        n = cur.fetchone()[0]
        print(f"  {t:<25} {n:>12,} rows")
conn.close()
COPY vs INSERT performance:
INSERT 8M rows: ~2–3 hours
COPY 8M rows: ~5–10 minutes

COPY bypasses row-by-row parsing, logging, and constraint checking during load. It's the industry-standard method for bulk data ingestion.
If COPY fails with a permission error — PostgreSQL's COPY FROM reads as the postgres OS user, not your Python process. Fix it by using copy_expert instead which streams from Python:

cur.copy_expert(f"COPY {table}({cols}) FROM STDIN WITH CSV HEADER", open(csv_path))

This works regardless of OS file permissions.

File: sql/queries.sql

SQL — sql/queries.sql
-- Q1: TOP INVENTORS — who has filed the most patents?
SELECT
    inv.name,
    inv.country,
    COUNT(pi.patent_id) AS patent_count
FROM inventors inv
JOIN patent_inventor pi ON inv.inventor_id = pi.inventor_id
GROUP BY inv.inventor_id, inv.name, inv.country
ORDER BY patent_count DESC
LIMIT 20;


-- Q2: TOP COMPANIES — which assignees own the most patents?
SELECT
    c.name,
    COUNT(pa.patent_id) AS patent_count
FROM companies c
JOIN patent_assignee pa ON c.company_id = pa.company_id
GROUP BY c.company_id, c.name
ORDER BY patent_count DESC
LIMIT 20;


-- Q3: TOP COUNTRIES — by number of patents filed
SELECT
    inv.country,
    COUNT(pi.patent_id) AS patent_count,
    ROUND(
        COUNT(pi.patent_id) * 100.0 /
        SUM(COUNT(pi.patent_id)) OVER(), 2
    ) AS share_pct
FROM inventors inv
JOIN patent_inventor pi ON inv.inventor_id = pi.inventor_id
WHERE inv.country NOT IN ('Unknown', '')
GROUP BY inv.country
ORDER BY patent_count DESC
LIMIT 20;


-- Q4: TRENDS OVER TIME — annual patent filing volume
SELECT
    year,
    COUNT(*) AS patents_filed,
    SUM(COUNT(*)) OVER (ORDER BY year) AS running_total
FROM patents
WHERE year BETWEEN 1976 AND 2024
GROUP BY year
ORDER BY year;


-- Q5: JOIN QUERY — patents enriched with inventor and company
SELECT
    p.patent_id,
    p.title,
    p.year,
    inv.name        AS inventor_name,
    inv.country,
    c.name          AS company_name
FROM patents p
JOIN patent_inventor pi  ON p.patent_id = pi.patent_id
JOIN inventors inv       ON pi.inventor_id = inv.inventor_id
LEFT JOIN patent_assignee pa ON p.patent_id = pa.patent_id
LEFT JOIN companies c    ON pa.company_id = c.company_id
WHERE p.year = 2023
LIMIT 5000;


-- Q6: CTE QUERY — top inventor per country (multi-step logic)
WITH inventor_totals AS (
    -- Step 1: count patents per inventor
    SELECT
        inv.inventor_id,
        inv.name,
        inv.country,
        COUNT(pi.patent_id) AS patent_count
    FROM inventors inv
    JOIN patent_inventor pi ON inv.inventor_id = pi.inventor_id
    WHERE inv.country NOT IN ('Unknown', '')
    GROUP BY inv.inventor_id, inv.name, inv.country
),
country_leaders AS (
    -- Step 2: rank inventors within each country
    SELECT
        *,
        ROW_NUMBER() OVER (
            PARTITION BY country
            ORDER BY patent_count DESC
        ) AS rn
    FROM inventor_totals
)
-- Step 3: keep only the top inventor from each country
SELECT name, country, patent_count
FROM country_leaders
WHERE rn = 1
ORDER BY patent_count DESC
LIMIT 30;


-- Q7: RANKING QUERY — inventors ranked with window functions
SELECT
    inv.name,
    inv.country,
    COUNT(pi.patent_id)                                       AS patent_count,
    RANK()        OVER (ORDER BY COUNT(pi.patent_id) DESC)   AS rank,
    DENSE_RANK()  OVER (ORDER BY COUNT(pi.patent_id) DESC)   AS dense_rank,
    PERCENT_RANK() OVER (ORDER BY COUNT(pi.patent_id) DESC)  AS pct_rank,
    NTILE(10)    OVER (ORDER BY COUNT(pi.patent_id) DESC)   AS decile
FROM inventors inv
JOIN patent_inventor pi ON inv.inventor_id = pi.inventor_id
GROUP BY inv.inventor_id, inv.name, inv.country
ORDER BY patent_count DESC
LIMIT 100;

import sys, os, json
import psycopg2
import psycopg2.extras
import pandas as pd
sys.path.insert(0, "scripts")
from config import get_conn

OUT = "reports"
os.makedirs(OUT, exist_ok=True)
conn = get_conn()


# ─── Helper: run query → pandas DataFrame ─────────────────────────────────────
def query_df(sql, limit=None):
    if limit:
        sql = sql.rstrip(";") + f" LIMIT {limit}"
    return pd.read_sql_query(sql, conn)


# ─── Load results ──────────────────────────────────────────────────────────────
top_inventors = query_df("""
    SELECT inv.name, inv.country, COUNT(pi.patent_id) AS patents
    FROM inventors inv JOIN patent_inventor pi ON inv.inventor_id = pi.inventor_id
    GROUP BY inv.inventor_id, inv.name, inv.country
    ORDER BY patents DESC LIMIT 20
""")

top_companies = query_df("""
    SELECT c.name, COUNT(pa.patent_id) AS patents
    FROM companies c JOIN patent_assignee pa ON c.company_id = pa.company_id
    GROUP BY c.company_id, c.name ORDER BY patents DESC LIMIT 20
""")

top_countries = query_df("""
    SELECT inv.country,
           COUNT(pi.patent_id) AS patents,
           ROUND(COUNT(pi.patent_id)*100.0/SUM(COUNT(pi.patent_id)) OVER(), 2) AS share_pct
    FROM inventors inv JOIN patent_inventor pi ON inv.inventor_id = pi.inventor_id
    WHERE inv.country NOT IN ('Unknown','')
    GROUP BY inv.country ORDER BY patents DESC LIMIT 20
""")

year_trends = query_df("""
    SELECT year, COUNT(*) AS patents_filed
    FROM patents WHERE year BETWEEN 1976 AND 2024
    GROUP BY year ORDER BY year
""")

total_patents = pd.read_sql_query("SELECT COUNT(*) AS n FROM patents", conn).iloc[0]["n"]
conn.close()


# ─── A. Console Report ─────────────────────────────────────────────────────────

Common Errors & Fixes

❌ psycopg2 install fails

Use the binary version: pip install psycopg2-binary. If still failing on Linux, run sudo apt install libpq-dev first.

❌ COPY permission denied

PostgreSQL's server process can't read your user's files. Use copy_expert instead which streams data from Python's file handle — no file system permission needed.

❌ Column name mismatch

Run head -1 data/raw/g_patent.tsv | tr '\t' '\n' | cat -n to see the exact column names and update usecols= in 02_clean.py to match.

❌ FK violation on COPY

A patent_id in the relationship table doesn't exist in patents. This means your clean_patents.csv was limited (nrows) but the relationship file is full. Either remove the nrows limit or load relationships after filtering to only patent IDs that exist.

❌ MemoryError during cleaning

Reduce the chunk size from 100k to 50k: change CHUNK = 50_000 in 02_clean.py. The chunked approach should prevent this, but very low RAM machines may need smaller chunks.

❌ Queries timeout / very slow

Check that indexes exist: SELECT indexname FROM pg_indexes WHERE tablename='patent_inventor'; If missing, run the CREATE INDEX lines from schema.sql manually in psql.
